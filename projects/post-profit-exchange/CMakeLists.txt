cmake_minimum_required(VERSION 3.25)
project(post_profit_exchange VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS code/configs/exchange.cfg)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/code/configs/exchange.cfg" PH_EXCHANGE_DEFAULTS)
string(FIND "${PH_EXCHANGE_DEFAULTS}" ")PHXCFG\"" PH_EXCHANGE_DELIMITER)
if(NOT PH_EXCHANGE_DELIMITER EQUAL -1)
  message(FATAL_ERROR "Configuration contains reserved C++ embedding delimiter")
endif()
configure_file(code/default_config.hpp.in generated/default_config.hpp @ONLY)
if(NOT TARGET ph_price_enumeration)
  add_subdirectory(../../tools/price-optimization price-optimization)
endif()
if(NOT TARGET ph::exponential_smoothing)
  add_subdirectory(../../tools/exponential-smoothing exponential-smoothing)
endif()
add_library(ph_exchange_models code/models.cpp)
add_library(ph::exchange_models ALIAS ph_exchange_models)
target_include_directories(ph_exchange_models PUBLIC code)
target_include_directories(ph_exchange_models SYSTEM PUBLIC ../../tools/price-optimization/third_party)
target_link_libraries(ph_exchange_models PUBLIC ph::exponential_smoothing)
add_library(ph_exchange_simulation code/simulation.cpp)
target_include_directories(ph_exchange_simulation PUBLIC code)
target_include_directories(ph_exchange_simulation PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(ph_exchange_simulation SYSTEM PUBLIC ../../tools/price-optimization/third_party)
target_link_libraries(ph_exchange_simulation PUBLIC ph::price_enumeration ph::exchange_models)
if(MSVC)
  target_compile_options(ph_exchange_simulation PRIVATE /W4)
else()
  target_compile_options(ph_exchange_simulation PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(exchange_simulation code/main.cpp code/files.cpp)
target_include_directories(exchange_simulation PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(exchange_simulation PRIVATE ph_exchange_simulation)
if(BUILD_TESTING)
  add_executable(exchange_files_test code/tests/files_test.cpp code/files.cpp)
  target_include_directories(exchange_files_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_definitions(exchange_files_test PRIVATE PH_EXCHANGE_SOURCE_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/code/configs/exchange.cfg")
  target_link_libraries(exchange_files_test PRIVATE ph_exchange_simulation)
  add_test(NAME exchange_files COMMAND exchange_files_test)
  add_executable(exchange_models_test code/tests/models_test.cpp)
  target_link_libraries(exchange_models_test PRIVATE ph::exchange_models)
  add_test(NAME exchange_models COMMAND exchange_models_test)
  add_executable(exchange_simulation_test code/tests/simulation_test.cpp)
  target_include_directories(exchange_simulation_test SYSTEM PRIVATE ../../tools/price-optimization/third_party)
  target_link_libraries(exchange_simulation_test PRIVATE ph_exchange_simulation)
  add_test(NAME exchange_simulation COMMAND exchange_simulation_test)
  set_tests_properties(exchange_simulation PROPERTIES TIMEOUT 30)
endif()

# The environment is a native consumer; it does not enter the WASM simulator.
option(PH_EXCHANGE_BUILD_ENVIRONMENT "Build native Linux environment commands using IdyDB" ON)
if(PH_EXCHANGE_BUILD_ENVIRONMENT)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Environment commands currently target Linux; disable PH_EXCHANGE_BUILD_ENVIRONMENT for simulator-only builds")
  endif()
  if(NOT TARGET ph::idydb)
    add_subdirectory(../../tools/idydb idydb)
  endif()
  add_library(ph_exchange_environment
    code/operations/opening.cpp
    code/operations/transactions.cpp
    code/storage/environment_store.cpp
    code/interfaces/environment.cpp)
  target_include_directories(ph_exchange_environment PUBLIC code)
  target_include_directories(ph_exchange_environment SYSTEM PUBLIC ../../tools/price-optimization/third_party)
  target_link_libraries(ph_exchange_environment PRIVATE ph::idydb)
  target_compile_options(ph_exchange_environment PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(exchange_environment code/interfaces/environment_main.cpp)
  target_link_libraries(exchange_environment PRIVATE ph_exchange_environment)
  if(BUILD_TESTING)
    add_executable(exchange_environment_store_test code/tests/environment_store_test.cpp)
    target_link_libraries(exchange_environment_store_test PRIVATE ph_exchange_environment ph::idydb)
    add_test(NAME exchange_environment_store COMMAND exchange_environment_store_test)
    add_executable(exchange_transactions_test code/tests/transactions_test.cpp)
    target_link_libraries(exchange_transactions_test PRIVATE ph_exchange_environment)
    add_test(NAME exchange_transactions COMMAND exchange_transactions_test)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME exchange_transactions_integration
      COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/code/tests/transactions_integration_test.py"
        $<TARGET_FILE:exchange_environment> "${CMAKE_CURRENT_SOURCE_DIR}")
    set_tests_properties(exchange_transactions_integration PROPERTIES TIMEOUT 60)
    add_executable(exchange_environment_test code/tests/environment_test.cpp)
    target_link_libraries(exchange_environment_test PRIVATE ph_exchange_environment)
    add_test(NAME exchange_environment COMMAND exchange_environment_test)
    set_tests_properties(exchange_environment_store exchange_environment PROPERTIES TIMEOUT 30)
  endif()
endif()
